cmake_minimum_required(VERSION 3.13)


# compile-time options

# Version numbers and build options now defined in root CMakeLists.txt

# end compile-time options


set(PROGRAM ${FIRMWARE_FILE})

add_executable(${PROGRAM} )

target_sources(${PROGRAM} PRIVATE
  main.c
  palette.c
  clocks.c
  renderer.c
  livetest.c
  tms_bus.c
  config.c
  flash.c
  gpio.c
  temperature.c
  clocks.pio.h
  tms9918.pio.h
)

pico_set_program_name(${PROGRAM} "pico9918")
pico_set_program_version(${PROGRAM} ${PICO9918_VERSION})
pico_set_program_description(${PROGRAM} "PICO9918 VDP")
pico_set_program_url(${PROGRAM} "https://github.com/visrealm/pico9918")

# The overlay image assets (splash + font) are generated by the library now -
# see core/pico9918_image_assets.cmake. Both are consumed
# entirely inside the library.

# generate header file from pio
pico_generate_pio_header(${PROGRAM} ${CMAKE_CURRENT_LIST_DIR}/clocks.pio)
pico_generate_pio_header(${PROGRAM} ${CMAKE_CURRENT_LIST_DIR}/tms9918.pio)

# Modify the below lines to enable/disable output over UART/USB
pico_enable_stdio_uart(${PROGRAM} 0)
pico_enable_stdio_usb(${PROGRAM} 0)

pico_set_binary_type(${PROGRAM} copy_to_ram)

# Additive linker fragment: adds a .flashcode output section ahead of .text.
# Always linked - it's an empty, harmless section unless PICO9918_COLD_IN_FLASH
# is on, in which case __in_flash_func() (see xip.h) routes functions into it.
target_link_options(${PROGRAM} PRIVATE "LINKER:--script=${CMAKE_CURRENT_LIST_DIR}/xip_sections.ld")
set_property(TARGET ${PROGRAM} APPEND PROPERTY LINK_DEPENDS ${CMAKE_CURRENT_LIST_DIR}/xip_sections.ld)

# A handful of vendored Pico SDK functions (boot-only clock bring-up) are
# also safe to move into flash, but we can't annotate their source with
# __in_flash_func() - see xip_sdk_sections.ld for what and why. Unlike the
# fragment above, the SDK source has no PICO9918_COLD_IN_FLASH awareness to
# gate itself with, so this one is only linked in when the option is on.
if (PICO9918_COLD_IN_FLASH)
  target_link_options(${PROGRAM} PRIVATE "LINKER:--script=${CMAKE_CURRENT_LIST_DIR}/xip_sdk_sections.ld")
  set_property(TARGET ${PROGRAM} APPEND PROPERTY LINK_DEPENDS ${CMAKE_CURRENT_LIST_DIR}/xip_sdk_sections.ld)
endif()

# Bounds each scratch bank against the stack sharing it - ld checks neither
# against the .scratch_* data, and does not track SCRATCH_Y at all.
target_link_options(${PROGRAM} PRIVATE "LINKER:--script=${CMAKE_CURRENT_LIST_DIR}/scratch_guard.ld")
set_property(TARGET ${PROGRAM} APPEND PROPERTY LINK_DEPENDS ${CMAKE_CURRENT_LIST_DIR}/scratch_guard.ld)

target_compile_definitions(${PROGRAM} PRIVATE
    PICO9918_ENABLE_SCART=$<BOOL:${PICO9918_ENABLE_SCART}>
    PICO9918_DIAG=$<BOOL:${PICO9918_DIAG}>
    PICO9918_GPU_FRAME_COUNTER=$<BOOL:${PICO9918_GPU_FRAME_COUNTER}>
    PICO9918_COLD_IN_FLASH=$<BOOL:${PICO9918_COLD_IN_FLASH}>
    PICO9918_VERSION="${PICO9918_VERSION}"
    PICO9918_MAJOR_VER=${PICO9918_MAJOR_VER}
    PICO9918_MINOR_VER=${PICO9918_MINOR_VER}
    PICO9918_PATCH_VER=${PICO9918_PATCH_VER}
    PICO_DISABLE_SHARED_IRQ_HANDLERS=1
    PICO_PANIC_FUNCTION=
    # both stacks share the scratch banks with render tables and buffers
    PICO_STACK_SIZE=1024
    PICO_CORE1_STACK_SIZE=1024
    # We only ever use sleep_ms() for boot-time GPIO/PLL settle delays (never
    # add_alarm_at/add_repeating_timer), so the default alarm pool's IRQ+WFE
    # wake machinery is dead weight - drop it and let sleep_ms() busy-wait.
    PICO_TIME_DEFAULT_ALARM_POOL_DISABLED=1
)


target_link_libraries(${PROGRAM} PUBLIC
        pico_stdlib
        pico_multicore
        hardware_dma
        hardware_interp
        hardware_pio
        hardware_adc
        hardware_flash
        pico9918-vga
        pico9918_core)

# pico9918_core is a portable library (also builds standalone/desktop) so this
# stays here rather than in its own CMakeLists.txt - see PICO9918_IN_FLASH_FUNC
# in pico9918.c.
target_compile_definitions(pico9918_core PRIVATE
    PICO9918_COLD_IN_FLASH=$<BOOL:${PICO9918_COLD_IN_FLASH}>
    PICO_TIME_DEFAULT_ALARM_POOL_DISABLED=1)

# Set the executable suffix to .elf for all platforms
set_target_properties(${PROGRAM} PROPERTIES SUFFIX .elf)

pico_add_extra_outputs(${PROGRAM})

# Copy final artifacts to dist directory for unified output
add_custom_command(
    OUTPUT ${CMAKE_BINARY_DIR}/dist/${PROGRAM}.uf2
    COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_BINARY_DIR}/dist
    COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE_DIR:${PROGRAM}>/${PROGRAM}.uf2 ${CMAKE_BINARY_DIR}/dist/
    DEPENDS ${PROGRAM}
    COMMENT "Copying ${PROGRAM}.uf2 to dist/"
)

# Create a target for the dist uf2 file
add_custom_target(${PROGRAM}_dist DEPENDS ${CMAKE_BINARY_DIR}/dist/${PROGRAM}.uf2)

# Create a generic firmware target for easier building
add_custom_target(firmware DEPENDS ${CMAKE_BINARY_DIR}/dist/${PROGRAM}.uf2)

add_subdirectory(pio-utils)
add_subdirectory(vga)
